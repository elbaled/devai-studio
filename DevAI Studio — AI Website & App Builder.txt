// Compute line-by-line diff between two file strings
function computeLineDiff(oldText = '', newText = '') {
  const oldLines = oldText.split('\n');
  const newLines = newText.split('\n');
  const maxLen = Math.max(oldLines.length, newLines.length);
  const diff = [];

  let i = 0;
  let j = 0;
  while (i < oldLines.length || j < newLines.length) {
    if (i < oldLines.length && j < newLines.length && oldLines[i] === newLines[j]) {
      diff.push({ type: 'same', oldLine: i + 1, newLine: j + 1, text: oldLines[i] });
      i++;
      j++;
    } else if (j < newLines.length && !oldLines.slice(i, i + 4).includes(newLines[j])) {
      diff.push({ type: 'add', oldLine: null, newLine: j + 1, text: newLines[j] });
      j++;
    } else if (i < oldLines.length) {
      diff.push({ type: 'remove', oldLine: i + 1, newLine: null, text: oldLines[i] });
      i++;
    } else {
      break;
    }
    if (diff.length > 800) break; // safety cap
  }
  return diff;
}

// Static code diagnostics & security linter
function lintFileContent(filePath, content = '') {
  const issues = [];
  if (!content) return issues;

  const lines = content.split('\n');

  if (filePath.endsWith('.js') || filePath.endsWith('.jsx') || filePath.endsWith('.ts') || filePath.endsWith('.tsx')) {
    lines.forEach((line, idx) => {
      if (line.includes('eval(')) {
        issues.push({
          line: idx + 1,
          severity: 'error',
          message: 'تحذير أمني خطير: استخدام eval() يعرض التطبيق لهجمات حقن الأكواد (Code Injection).'
        });
      }
      if (line.match(/AIza[0-9A-Za-z-_]{35}/) && !filePath.includes('firebase')) {
        issues.push({
          line: idx + 1,
          severity: 'warning',
          message: 'تنبيه أمني: يبدو أن هناك مفتاح API مكتوباً بشكل مباشر في الكود الأمامي.'
        });
      }
      if (line.includes('innerHTML') && !line.includes('escape') && !line.includes('`')) {
        issues.push({
          line: idx + 1,
          severity: 'info',
          message: 'ملاحظة أمان: تأكد من تنقية مدخلات المستخدم قبل تمريرها إلى innerHTML لتجنب ثغرات XSS.'
        });
      }
    });

    // Check bracket balance for pure JS files
    if (filePath.endsWith('.js')) {
      const openBraces = (content.match(/\{/g) || []).length;
      const closeBraces = (content.match(/\}/g) || []).length;
      if (openBraces !== closeBraces) {
        issues.push({
          line: lines.length,
          severity: 'error',
          message: `عدم تطابق في الأقواس المعقوفة: فتح (${openBraces}) مقابل إغلاق (${closeBraces}).`
        });
      }
    }
  }

  if (filePath.endsWith('.json')) {
    try {
      JSON.parse(content);
    } catch (err) {
      issues.push({
        line: 1,
        severity: 'error',
        message: `خطأ في صيغة JSON: ${err.message}`
      });
    }
  }

  if (filePath.endsWith('.rules')) {
    lines.forEach((line, idx) => {
      if (line.replace(/\s+/g, '').includes('allowread,write:iftrue') || line.replace(/\s+/g, '').includes('allowwrite:iftrue')) {
        issues.push({
          line: idx + 1,
          severity: 'error',
          message: 'ثغرة أمنية حرجة: القاعدة تسمح لأي شخص بتعديل أو قراءة قاعدة البيانات بالكامل دون مصادقة!'
        });
      }
    });
  }

  return issues;
}

// Load JSZip dynamically from CDN for real .zip export & import
async function loadJSZipLibrary() {
  if (window.JSZip) return window.JSZip;
  return new Promise((resolve, reject) => {
    const script = document.createElement('script');
    script.src = 'https://cdnjs.cloudflare.com/ajax/libs/jszip/3.10.1/jszip.min.js';
    script.onload = () => resolve(window.JSZip);
    script.onerror = () => reject(new Error('تعذر تحميل مكتبة ضغط الملفات JSZip'));
    document.head.appendChild(script);
  });
}

// Build isolated sandbox HTML combining project HTML, CSS, and JS files + Console Interceptor
function buildSandboxedPreviewHtml(files) {
  let html = files['index.html'] || files['public/index.html'] || '';

  // If project only has React/Next.js files without index.html, synthesize an interactive preview wrapper
  if (!html) {
    const firstFile = Object.keys(files)[0] || '';
    return `<!DOCTYPE html>
<html lang="ar" dir="rtl">
<head>
  <meta charset="UTF-8" />
  <script src="https://cdn.tailwindcss.com"></script>
</head>

    <p class="text-sm text-slate-300 leading-relaxed mb-4">
      يحتوي هذا المشروع على ملفات تحتاج إلى بيئة بناء (مثل Next.js / TypeScript / Node.js) أو لا يتضمن ملف <code>index.html</code> قياسي في الجذر.
    </p>
    <p class="text-xs text-slate-400">
      يمكنك طلب إضافة ملف <code>index.html</code> للمعاينة الفورية من المساعد الذكي، أو فتح "دليل تشغيل البنية والخادم" في أعلى المحرر.
    </p>
  </div>
</body>
</html>`;
  }

  // Inject all CSS files inline
  let combinedCss = '';
  Object.entries(files).forEach(([path, content]) => {
    if (path.endsWith('.css')) {
      combinedCss += `\n/* === File: ${path} === */\n${content}\n`;
    }
  });

  // Inject all JS files inline (excluding service workers or server configs)
  let combinedJs = '';
  Object.entries(files).forEach(([path, content]) => {
    if (
      path.endsWith('.js') &&
      !path.includes('config') &&
      !path.endsWith('sw.js') &&
      !path.includes('server')
    ) {
      combinedJs += `\n// === File: ${path} ===\ntry {\n${content}\n} catch (err) {\n  console.error("خطأ في ملف ${path}: " + err.message);\n}\n`;
    }
  });

  // Console & Runtime Error Bridge to Parent IDE
  const consoleBridgeScript = `
<script>
(function() {
  const sendToIde = (level, args) => {
    try {
      const text = Array.from(args).map(a => {
        if (typeof a === 'object') {
          try { return JSON.stringify(a); } catch (e) { return String(a); }
        }
        return String(a);
      }).join(' ');
      window.parent.postMessage({ source: 'devai-sandbox', level, text, time: new Date().toLocaleTimeString() }, '*');
    } catch (e) {}
  };
  const origLog = console.log;
  const origWarn = console.warn;
  const origError = console.error;
  console.log = function() { sendToIde('info', arguments); origLog.apply(console, arguments); };
  console.warn = function() { sendToIde('warn', arguments); origWarn.apply(console, arguments); };
  console.error = function() { sendToIde('error', arguments); origError.apply(console, arguments); };
  window.addEventListener('error', function(e) {
    sendToIde('error', [e.message + (e.filename ? ' (' + e.lineno + ':' + e.colno + ')' : '')]);
  });
})();
</script>`;

  // Remove external relative script/style tags that we already bundle inline
  let processedHtml = html
    .replace(/<link[^>]+href=["'](?!https?:\/\/)[^"']+\.css["'][^>]*>/gi, '')
    .replace(/<script[^>]+src=["'](?!https?:\/\/)[^"']+\.js["'][^>]*><\/script>/gi, '');

  if (processedHtml.includes('</head>')) {
    processedHtml = processedHtml.replace(
      '</head>',
      `${consoleBridgeScript}\n<style>${combinedCss}</style>\n</head>`
    );
  } else {
    processedHtml = `${consoleBridgeScript}<style>${combinedCss}</style>` + processedHtml;
  }

  if (processedHtml.includes('</body>')) {
    processedHtml = processedHtml.replace(
      '</body>',
      `<script>${combinedJs}</script>\n</body>`
    );
  } else {
    processedHtml += `<script>${combinedJs}</script>`;
  }

  return processedHtml;
}

// 5. REAL GEMINI AI ARCHITECT ENGINE (WITH EXPONENTIAL BACKOFF & STRUCTURED JSON)
// ============================================================================
async function callGeminiArchitect({
  userPrompt,
  projectFiles,
  projectName,
  chatHistory,
  customApiKey = ''
}) {
  const apiKey = customApiKey.trim() || '';
  const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

  // Summarize existing project files so AI has full context and modifies surgically
  const fileContextString = Object.entries(projectFiles)
    .map(([path, content]) => `--- FILE: ${path} ---\n${content}\n`)
    .join('\n');

  const recentConversation = chatHistory
    .slice(-5)
    .map(m => `${m.role === 'user' ? 'المستخدم' : 'المساعد'}: ${m.text}`)
    .join('\n');

  const systemInstruction = `أنت "مهندس DevAI Studio"، خبير عالمي في هندسة البرمجيات وبناء مواقع الويب والتطبيقات (HTML5, Tailwind CSS, JavaScript, React, Next.js, TypeScript, PWA, Android Capacitor, Firebase v10, Security Rules, APIs).

القواعد الهندسية الصارمة:
1. اقرأ ملفات المشروع الحالية بعناية وحافظ على السياق والميزات الموجودة.
2. إذا طلب المستخدم تعديلاً بسيطاً أو إضافة ميزة، قم بتعديل الملفات المعنية فقط أو إنشاء ملفات جديدة مترابطة دون إفساد الخصائص السابقة.
3. تأكد دائماً من أن ملف "index.html" وملفات "styles/main.css" و "scripts/app.js" مترابطة وتعمل مباشرة في المعاينة الحية بتصميم عربي عصري وجذاب باستخدام Tailwind CSS.
4. لا تستخدم أبداً بيانات وهمية فارغة أو عبارات مثل "أكمل الكود بنفسك"؛ اكتب أكواداً كاملة وجاهزة للتشغيل الفعلي.
5. اشرح في حقل explanation باللغة العربية: الخطة الهندسية، الملفات التي قمت بإنشائها أو تعديلها، وأي نصائح للأمان أو الأداء.
6. أرجع النتيجة حصرياً بصيغة JSON مطابقة للـ Schema المحددة.`;

  const fullUserPrompt = `اسم المشروع الحالي: ${projectName}

سجل المحادثة الأخير:
${recentConversation}

ملفات المشروع الحالية:
${fileContextString}

طلب المستخدم الجديد:
${userPrompt}

يرجى تحليل الطلب، ووضع خطة واضحة في الشرح، وتوليد التحديثات اللازمة للملفات.`;

  const payload = {
    contents: [{ role: 'user', parts: [{ text: fullUserPrompt }] }],
    systemInstruction: { parts: [{ text: systemInstruction }] },
    generationConfig: {
      responseMimeType: 'application/json',
      responseSchema: {
        type: 'OBJECT',
        properties: {
          explanation: {
            type: 'STRING',
            description: 'شرح هندسي وافٍ باللغة العربية للخطة والتعديلات وملاحظات الأمان أو الأداء'
          },
          files: {
            type: 'ARRAY',
            description: 'قائمة الملفات التي يجب إنشاؤها أو تحديثها أو حذفها فقط',
            items: {
              type: 'OBJECT',
              properties: {
                path: { type: 'STRING', description: 'مسار الملف مثل index.html أو scripts/app.js' },
                action: { type: 'STRING', description: 'create أو update أو delete' },
                summary: { type: 'STRING', description: 'وصف مختصر بالعربية لما تم عمله في هذا الملف' },
                content: { type: 'STRING', description: 'الكود الكامل النهائي للملف بدون أي اختصار' }
              },
              required: ['path', 'action', 'summary', 'content']
            }
          }
        },
        required: ['explanation', 'files']
      }
    }
  };

  // Exponential backoff retry (1s, 2s, 4s)
  const delays = [1000, 2000, 4000];
  let lastError = null;

  for (let attempt = 0; attempt <= delays.length; attempt++) {
    try {
      const response = await fetch(apiUrl, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });

      if (!response.ok) {
        const errText = await response.text();
        throw new Error(`HTTP ${response.status}: ${errText}`);
      }

      const result = await response.json();
      const rawJson = result?.candidates?.[0]?.content?.parts?.[0]?.text;
      if (!rawJson) {
        throw new Error('لم يتم استلام استجابة نصية صالحة من نموذج الذكاء الاصطناعي.');
      }

      return JSON.parse(rawJson);
    } catch (err) {
      lastError = err;
      if (attempt < delays.length) {
        await new Promise(r => setTimeout(r, delays[attempt]));
      }
    }
  }

  throw lastError;
}

                  <option value="pwa-android">تطبيق PWA + Capacitor Android</option>
                  <option value="react-next">مشروع Next.js + TypeScript + Tailwind</option>
                </select>
              </div>
              <div className="flex justify-end gap-2 pt-2">
                <button
                  type="button"
                  onClick={() => setShowNewProjectModal(false)}
                  className="px-4 py-2 rounded-xl bg-slate-800 text-slate-300 text-xs"
                >
                  إلغاء
                </button>
                <button
                  type="submit"
                  className="px-4 py-2 rounded-xl bg-indigo-600 hover:bg-indigo-500 text-white text-xs font-bold"
                >
                  إنشاء المشروع الآن
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* =================================================================== */}
      {/* MODAL 2: API KEYS & SECURE SERVER ARCHITECTURE CONFIG               */}
      {/* =================================================================== */}
      {showApiConfigModal && (
        <div className="fixed inset-0 z-50 bg-black/75 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-slate-900 border border-slate-800 rounded-2xl max-w-lg w-full p-6 shadow-2xl space-y-4">
            <div className="flex items-center justify-between">
              <h3 className="text-base font-bold text-white flex items-center gap-2">
                <Key className="w-5 h-5 text-indigo-400" />
                <span>إدارة مفاتيح API واتصال الذكاء الاصطناعي الآمن</span>
              </h3>
              <button onClick={() => setShowApiConfigModal(false)} className="text-slate-400 hover:text-white">
                ✕
              </button>
            </div>

            <div className="p-3.5 rounded-xl bg-indigo-950/40 border border-indigo-500/30 text-xs text-slate-200 leading-relaxed">
              <strong>حالة الاتصال الحالية:</strong> المنصة متصلة تلقائياً بواجهة <code>Gemini 2.5 Flash</code> المدمجة في بيئة العمل.
              <br />
              في حال رغبتك بتشغيل المنصة على خادمك الخاص أو استخدام مفتاح خاص للجلسة الحالية، يمكنك إدخاله بالأسفل (لا يتم تخزينه في الكود المصدري للمشروع أبداً).
            </div>

            <div>
              <label className="block text-xs text-slate-300 mb-1">
                مفتاح Gemini API اختياري للجلسة الحالية (يُترك فارغاً لاستخدام اتصال المنصة التلقائي):
              </label>
              <input
                type="password"
                value={customGeminiKey}
                onChange={e => setCustomGeminiKey(e.target.value)}
                placeholder="AIzaSy... (اختياري)"
                dir="ltr"
                className="w-full px-3 py-2 rounded-xl bg-slate-950 border border-slate-800 text-xs text-white"
              />
            </div>

            <div className="p-3 rounded-xl bg-slate-950 border border-slate-800 text-[11px] text-slate-400 space-y-1">
              <div className="font-bold text-cyan-400">كيف تحمي مفاتيح API عند نشر منصتك؟</div>
              <p>
                راجع ملف <code>README.md</code> داخل المشروع؛ حيث قمنا بتوثيق طريقة إعداد خادم وسيط (Next.js API Route) يقرأ المفتاح من متغيرات البيئة <code>process.env.GEMINI_API_KEY</code> دون كشفه في الواجهة الأمامية.
              </p>
            </div>

            <div className="flex justify-end">
              <button
                onClick={() => {
                  setShowApiConfigModal(false);
                  notify('تم حفظ إعدادات الاتصال بنجاح!', 'success');
                }}
                className="px-5 py-2 rounded-xl bg-indigo-600 hover:bg-indigo-500 text-white text-xs font-bold"
              >
                حفظ وإغلاق
              </button>
            </div>
          </div>
        </div>
      )}

      {/* =================================================================== */}
      {/* MODAL 3: SERVER & BUILD RUNNER GUIDE                                */}
      {/* =================================================================== */}
      {showRunnerGuideModal && (
        <div className="fixed inset-0 z-50 bg-black/75 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-slate-900 border border-slate-800 rounded-2xl max-w-xl w-full p-6 shadow-2xl space-y-4">
            <div className="flex items-center justify-between">
              <h3 className="text-base font-bold text-white flex items-center gap-2">
                <Rocket className="w-5 h-5 text-amber-400" />
                <span>دليل تشغيل البنية التقنية والنشر (Build & Deploy)</span>
              </h3>
              <button onClick={() => setShowRunnerGuideModal(false)} className="text-slate-400 hover:text-white">
                ✕
              </button>
            </div>

            <div className="space-y-3 text-xs text-slate-300">
              <div className="p-3 rounded-xl bg-slate-950 border border-slate-800">
                <div className="font-bold text-indigo-400 mb-1">1. تشغيل مشاريع Next.js / React محلياً:</div>
                <pre className="font-mono text-[11px] text-emerald-300 overflow-x-auto" dir="ltr">
                  {`npm install\nnpm run dev`}
                </pre>
              </div>

              <div className="p-3 rounded-xl bg-slate-950 border border-slate-800">
                <div className="font-bold text-cyan-400 mb-1">2. تحويل المشروع إلى تطبيق Android APK:</div>
                <pre className="font-mono text-[11px] text-emerald-300 overflow-x-auto" dir="ltr">
                  {`npm install @capacitor/core @capacitor/cli @capacitor/android\nnpx cap add android\nnpx cap open android`}
                </pre>
              </div>

              <div className="p-3 rounded-xl bg-slate-950 border border-slate-800">
                <div className="font-bold text-amber-400 mb-1">3. النشر المباشر على Firebase Hosting:</div>
                <pre className="font-mono text-[11px] text-emerald-300 overflow-x-auto" dir="ltr">
                  {`npm install -g firebase-tools\nfirebase login\nfirebase deploy --only hosting,firestore:rules`}
                </pre>
              </div>
            </div>

            <div className="flex justify-end">
              <button
                onClick={() => setShowRunnerGuideModal(false)}
                className="px-4 py-2 rounded-xl bg-slate-800 hover:bg-slate-700 text-white text-xs font-semibold"
              >
                إغلاق
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
